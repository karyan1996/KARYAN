        const CORS_PROXY = 'https://api.allorigins.win/raw?url=';

        

        async function fetchWithProxy(url) {

            const response = await fetch(CORS_PROXY + encodeURIComponent(url));

            if (!response.ok) throw new Error('Netzwerkfehler');

            return response.text();

        }

        

        async function getUserId(username) {

            const html = await fetchWithProxy(`https://www.instagram.com/${username}/`);

            

            // Extrahiere das JSON aus dem HTML

            const match = html.match(/<script type="text\/javascript">window\._sharedData = (.*?);<\/script>/);

            if (!match) throw new Error('Profil nicht gefunden oder privat');

            

            const data = JSON.parse(match[1]);

            const user = data.entry_data.ProfilePage[0].graphql.user;

            

            if (user.is_private) throw new Error('Dieses Profil ist privat');

            

            return {

                id: user.id,

                username: user.username,

                fullName: user.full_name,

                followerCount: user.edge_followed_by.count,

                followingCount: user.edge_follow.count,

                profilePic: user.profile_pic_url

            };

        }

        

        async function getAllFollowers(userId, totalCount) {

            const allFollowers = [];

            let hasNextPage = true;

            let endCursor = '';

            

            while (hasNextPage) {

                const variables = {

                    id: userId,

                    include_reel: false,

                    fetch_mutual: false,

                    first: 50,

                    after: endCursor

                };

                

                const url = `https://www.instagram.com/graphql/query/?query_hash=37479f2b8209594dde7facb0d904896a&variables=${encodeURIComponent(JSON.stringify(variables))}`;

                

                const jsonText = await fetchWithProxy(url);

                const data = JSON.parse(jsonText);

                

                if (!data.data || !data.data.user) {

                    throw new Error('Instagram hat die Anfrage blockiert. Bitte warte einige Minuten.');

                }

                

                const edges = data.data.user.edge_followed_by.edges;

                

                edges.forEach(edge => {

                    allFollowers.push({

                        username: edge.node.username,

                        fullName: edge.node.full_name || '',

                        profilePic: edge.node.profile_pic_url || '',

                        isVerified: edge.node.is_verified || false

                    });

                });

                

                // Update Fortschritt

                const percent = Math.round((allFollowers.length / totalCount) * 100);

                updateProgress(allFollowers.length, totalCount, percent);

                

                hasNextPage = data.data.user.edge_followed_by.page_info.has_next_page;

                endCursor = data.data.user.edge_followed_by.page_info.end_cursor;

                

                // Pause zwischen Anfragen

                if (hasNextPage) {

                    await sleep(1500 + Math.random() * 1000);

                }

            }

            

            return allFollowers;

        }

        

        function sleep(ms) {

            return new Promise(resolve => setTimeout(resolve, ms));

        }

        function saveScan(username, data) {

            const key = `instatracker_${username}`;

            localStorage.setItem(key, JSON.stringify(data));

            

            // History aktualisieren

            let history = JSON.parse(localStorage.getItem('instatracker_history') || '[]');

            history = history.filter(h => h.username !== username);

            history.unshift({

                username: username,

                timestamp: new Date().toISOString(),

                followerCount: data.followers.length

            });

            if (history.length > 50) history = history.slice(0, 50);

            localStorage.setItem('instatracker_history', JSON.stringify(history));

        }

        

        function getLastScan(username) {

            const key = `instatracker_${username}`;

            const data = localStorage.getItem(key);

            return data ? JSON.parse(data) : null;

        }

        

        function getHistory() {

        }

        async function startScan() {

            const username = document.getElementById('usernameInput').value.trim().replace('@', '');

            

            if (!username) {

                alert('Bitte gib einen Instagram-Nutzernamen ein.');

                return;

            }

            

            const btn = document.getElementById('scanBtn');

            btn.disabled = true;

            btn.textContent = '⏳ Scanne...';

            

            showLoading();

            

            try {

                // 1. Profil-Info holen

                const userInfo = await getUserId(username);

                

                document.getElementById('loadingText').textContent = 

                    `Profil gefunden: ${userInfo.followerCount.toLocaleString()} Follower`;

                await sleep(500);

                

                // 2. Alle Follower laden

                const followers = await getAllFollowers(userInfo.id, userInfo.followerCount);

                

                // 3. Scan-Daten erstellen

                const scanData = {

                    userInfo: userInfo,

                    followers: followers,

                    timestamp: new Date().toISOString()

                };

                

                // 4. Mit letztem Scan vergleichen

                const lastScan = getLastScan(username);

                let comparison = null;

                

                if (lastScan) {

                    const lastUsernames = new Set(lastScan.followers.map(f => f.username));

                    const currentUsernames = new Set(followers.map(f => f.username));

                    

                    const newFollowers = followers.filter(f => !lastUsernames.has(f.username));

                    const unfollowed = lastScan.followers.filter(f => !currentUsernames.has(f.username));

                    

                    comparison = {

                        isFirstScan: false,

                        newFollowers: newFollowers,

                        unfollowed: unfollowed,

                        stats: {

                            previousCount: lastScan.followers.length,

                            currentCount: followers.length,

                            newCount: newFollowers.length,

                            unfollowedCount: unfollowed.length

                        }

                    };

                } else {

                    comparison = {

                        isFirstScan: true,

                        newFollowers: [],

                        unfollowed: [],

                        stats: {

                            previousCount: 0,

                            currentCount: followers.length,

                            newCount: 0,

                            unfollowedCount: 0

                        }

                    };

                </div>

                    <div class="stat-label">Entfolgt</div>

                </div>

                <div class="stat stat-total">

                    <div class="stat-number">${stats.currentCount.toLocaleString()}</div>

                    <div class="stat-label">Gesamt</div>

                </div>

            let currentTab = 'new';

            

            document.getElementById('tabsContainer').innerHTML = `

                    🟢 Neu (${comparison.newFollowers.length})

                </div>

                <div class="tab" onclick="switchTab('unfollowed', ${JSON.stringify(comparison.unfollowed).replace(/"/g, '&quot;')})">

                    🔴 Entfolgt (${comparison.unfollowed.length})

                </div>

            // Liste

            renderFollowerList(comparison.newFollowers);

            

            // Tabs zum Fenster hinzufügen

            window._comparison = comparison;

        }

        

        function switchTab(tab) {

            const tabs = document.querySelectorAll('.tab');

            tabs.forEach(t => t.classList.remove('active'));

            event.target.classList.add('active');

            

            const data = tab === 'new' ? window._comparison.newFollowers : window._comparison.unfollowed;

            renderFollowerList(data);

        }

        

        function renderFollowerList(followers) {

            const container = document.getElementById('followerList');

            

            if (followers.length === 0) {

                container.innerHTML = '<div class="empty-state">Keine Einträge</div>';

                return;

            }

            

            container.innerHTML = followers.map(f => `

                <div class="follower-item" onclick="window.open('https://instagram.com/${f.username}', '_blank')">

                    <div class="avatar">${(f.fullName || f.username).charAt(0).toUpperCase()}</div>

                    <div class="follower-info">

                        <div class="follower-username">${f.username} ${f.isVerified ? '✓' : ''}</div>

                        <div class="follower-name">${f.fullName || ''}</div>

                    </div>

                </div>

            `).join('');

        }

        

        function updateHistoryList() {

            const history = getHistory();

            const container = document.getElementById('historyContainer');

            

            if (history.length === 0) {

                container.innerHTML = '<div class="empty-state">Noch keine Scans durchgeführt.</div>';

                return;

            }

            

            container.innerHTML = history.slice(0, 10).map(h => `

                <div class="history-item" onclick="document.getElementById('usernameInput').value='${h.username}'; startScan();">

                    <div class="history-info">

                        <h3>@${h.username}</h3>

                        <span>${new Date(h.timestamp).toLocaleDateString('de-DE')}</span>

                    </div>

                    <div class="history-count">${h.followerCount.toLocaleString()}</div>

                </div>

            `).join('');

        }

        

        // Initial History laden

        updateHistoryList();

        

        // Enter-Taste zum Scannen

        document.getElementById('usernameInput').addEventListener('keypress', function(e) {

            if (e.key === 'Enter') startScan();

        });
